Purpose

This page asks whether the fat body DEGs from the diet-by-infection experiment also overlap with independent phase-responsive genes from S. gregaria head and thorax transcriptomes.

The concern is biological rather than technical: fat bodies were sampled after animals experienced the diet and infection treatments, but the experimental context may also include a shift toward semi-isolated conditions. If some fat body DEGs are also known phase-transition genes, those genes should be treated as potentially density-state sensitive. They are not removed from the analysis; they are flagged so the infection and diet interpretation can be more careful.

In the comparative phase dataset, isolated or solitarious animals were the reference state. Therefore, a positive phase log2 fold-change means a gene is higher in crowded/gregarious animals, and a negative phase log2 fold-change means a gene is higher in isolated/solitarious animals.

This rendered page uses the fat body DEG scenario All samples. Experiment fat body sample displays use all experiment samples, including the current outlier flags.

latest_dir <- function(pattern) {
  dirs <- list.dirs(results_root, recursive = FALSE, full.names = TRUE)
  dirs <- dirs[grepl(pattern, basename(dirs))]
  if (length(dirs) == 0) return(NA_character_)
  dirs[which.max(file.info(dirs)$mtime)]
}

    NA_character_
  } else if (grepl("^/", path_param)) {

  } else {

read_phase_table <- function(path, tissue) {
  tab <- read.csv(path, check.names = FALSE)
  bad_names <- is.na(names(tab)) | names(tab) == ""
  names(tab)[bad_names] <- paste0("row_id", seq_len(sum(bad_names)))
  if (!"GeneID" %in% names(tab)) {
    tab$GeneID <- tab[[1]]
  }
  tab |>
    dplyr::transmute(
      gene_id = as.character(GeneID),
      tissue = tissue,
      phase_baseMean = baseMean,
      phase_log2FoldChange = log2FoldChange,
      phase_padj = padj,
      phase_significant = !is.na(padj) &
        padj < params$phase_alpha &
        abs(log2FoldChange) >= params$phase_lfc_threshold,
      phase_direction = dplyr::case_when(
        !phase_significant ~ "No phase DEG",
        log2FoldChange > 0 ~ "Gregarious-up",
        log2FoldChange < 0 ~ "Solitarious-up",
        TRUE ~ "No phase DEG"
      )
    )
}

parse_cds_gene_bridge <- function(path) {
  gff <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", stringsAsFactors = FALSE)
  names(gff)[1:9] <- c("seqid", "source", "type", "start", "end", "score",

    dplyr::filter(!is.na(protein_id), !is.na(gene_id)) |>
    dplyr::distinct()
}

read_featurecounts_gene_counts <- function(path, sample_id) {
  tab <- read.delim(path, comment.char = "#", check.names = FALSE)
  if ("Geneid" %in% names(tab)) {
    count_col <- names(tab)[ncol(tab)]
    out <- tibble::tibble(
      gene_id = as.character(tab[["Geneid"]]),
      count = as.integer(tab[[count_col]])
    )
  } else {
    tab <- read.delim(path, comment.char = "#", header = FALSE,
                      check.names = FALSE)
    if (ncol(tab) < 2) {
      stop("Count file must contain at least gene ID and count columns: ", path)
    }
    out <- tibble::tibble(
      gene_id = as.character(tab[[1]]),
      count = as.integer(tab[[ncol(tab)]])
    )
  }
  out <- out |>
    dplyr::filter(!is.na(gene_id), nzchar(gene_id)) |>
    dplyr::group_by(gene_id) |>
    dplyr::summarise(count = sum(count, na.rm = TRUE), .groups = "drop")
  names(out)[2] <- sample_id
  out
}

parse_head_thorax_reference_metadata <- function(reference_dir) {
  files <- list.files(
    reference_dir,
    pattern = "^SGRE-(HEAD|THOX)-(CRD|ISO)-[0-9]+_MERGE_featurecounts[.]txt$",
    full.names = TRUE
  )
  if (length(files) == 0) {
    return(tibble::tibble(
      sample_id = character(),
      count_file = character(),
      tissue = character(),
      phase = character(),
      rearing_condition = character(),
      replicate = integer(),
      source = character(),
      display_group = character(),
      reference_role = character(),
      sample_label = character(),
      sample_block = character()
    ))
  }

  parsed <- stringr::str_match(
    basename(files),
    "^SGRE-(HEAD|THOX)-(CRD|ISO)-([0-9]+)_MERGE_featurecounts[.]txt$"
  )
  tibble::tibble(
    sample_id = stringr::str_remove(basename(files), "_featurecounts[.]txt$"),
    count_file = files,
    tissue_code = parsed[, 2],
    rearing_code = parsed[, 3],
    replicate = as.integer(parsed[, 4])
  ) |>
    dplyr::filter(!is.na(tissue_code), !is.na(rearing_code)) |>
    dplyr::mutate(
      tissue = dplyr::case_when(
        tissue_code == "HEAD" ~ "Head",
        tissue_code == "THOX" ~ "Thorax",
        TRUE ~ tissue_code
      ),
      phase = dplyr::case_when(
        rearing_code == "CRD" ~ "Gregarious",
        rearing_code == "ISO" ~ "Solitarious",
        TRUE ~ rearing_code
      ),
      rearing_condition = dplyr::case_when(
        rearing_code == "CRD" ~ "Crowded",
        rearing_code == "ISO" ~ "Isolated",
        TRUE ~ rearing_code
      ),
      source = "Time-course head/thorax phase reference",
      display_group = paste0(
        tissue, ": ",
        dplyr::if_else(rearing_condition == "Crowded",
                       "crowded/gregarious", "isolated/solitarious")
      ),
      reference_role = paste(tissue, tolower(rearing_condition),
                             "rep", replicate),
      sample_label = paste0(substr(tissue, 1, 1), "-",
                            rearing_code, "-", replicate),
      sample_block = paste(tissue, "reference")
    ) |>
    dplyr::arrange(tissue, rearing_condition, replicate)
}

Enrichment and concordance tests

The overlap tests ask whether fat body DEGs are enriched for phase-DEG genes more than expected from the analyzed gene universe. The sign tests ask whether overlapping genes tend to move in the same direction as the phase signature or the opposite direction.

fisher_one <- function(sig_genes, phase_genes, universe) {
  sig_genes <- intersect(unique(sig_genes), universe)
  phase_genes <- intersect(unique(phase_genes), universe)
  overlap <- length(intersect(sig_genes, phase_genes))
  tab <- matrix(c(
    overlap,
    length(setdiff(sig_genes, phase_genes)),
    length(setdiff(phase_genes, sig_genes)),
    length(setdiff(universe, union(sig_genes, phase_genes)))
  ), nrow = 2)
  test <- fisher.test(tab, alternative = "greater")
  tibble::tibble(
    n_fat_degs = length(sig_genes),
    n_phase_genes = length(phase_genes),
    n_overlap = overlap,
    odds_ratio = unname(test$estimate),
    p_value = test$p.value
  )
}

sign_test_one <- function(same_direction) {
  same_direction <- same_direction[!is.na(same_direction)]
  if (length(same_direction) < 2) {
    return(tibble::tibble(n_overlap = length(same_direction),
                          n_same_direction = sum(same_direction),
                          n_opposite_direction = sum(!same_direction),
                          p_value = NA_real_))
  }
  test <- binom.test(sum(same_direction), length(same_direction), p = 0.5)
  tibble::tibble(n_overlap = length(same_direction),
                 n_same_direction = sum(same_direction),
                 n_opposite_direction = sum(!same_direction),
                 p_value = test$p.value)
}

Cross-tissue VST overview

This exploratory view combines the diet-infection fat body experiment, the two time-course fat body controls, and the time-course gregaria head/thorax phase-reference samples. The purpose is to ask where the present fat body samples sit relative to independent tissue and phase references. Because tissue identity and library context can dominate this kind of visualization, the PCA and heatmap should be read as a structure/QC view, not as a formal cross-tissue differential-expression test.

The heatmap includes all sample columns, but only the most variable shared genes are drawn as rows so that the plot remains readable. The complete combined VST matrix is written to the output folder for follow-up checks.

Gene-source partition

This section separates three evidence layers that could otherwise be mixed together in interpretation:

  • tissue-associated reference expression patterns, from the FAT/head/thorax VST reference panel;
  • phase-associated genes, from the independent head/thorax phase DEG lists;
  • diet/infection-associated genes, from the selected fat body DEG scenario (All samples).

The tissue layer is treated as an exploratory expression-pattern layer rather than a formal tissue DEG test, because the reference samples were not generated as a balanced replicated tissue-DEG experiment. A gene is called tissue-associated here when its mean VST differs by at least 2 across reference tissues.

if (exists("vst_cross_mat") && exists("cross_metadata") &&
    exists("phase_gene_all") && exists("phase_layered_sig")) {
  reference_tissue_metadata <- as.data.frame(cross_metadata) |>
    dplyr::filter(sample_block != "Experiment fat body") |>
    dplyr::mutate(tissue = as.character(tissue))

  reference_tissue_sample_ids <- intersect(
    reference_tissue_metadata$sample_id,
    colnames(vst_cross_mat)
  )

  reference_tissue_long <- as.data.frame(
    vst_cross_mat[, reference_tissue_sample_ids, drop = FALSE]
  ) |>
    tibble::rownames_to_column("gene_id") |>
    tidyr::pivot_longer(-gene_id, names_to = "sample_id",
                        values_to = "vst") |>
    dplyr::left_join(
      reference_tissue_metadata |>
        dplyr::select(sample_id, tissue),
      by = "sample_id"
    )

  reference_tissue_signal <- reference_tissue_long |>
    dplyr::group_by(gene_id, tissue) |>
    dplyr::summarise(mean_vst = mean(vst, na.rm = TRUE), .groups = "drop") |>
    dplyr::group_by(gene_id) |>
    dplyr::summarise(
      n_reference_tissues = dplyr::n_distinct(tissue),
      tissue_vst_range = diff(range(mean_vst, na.rm = TRUE)),
      highest_reference_tissue = tissue[which.max(mean_vst)][1],
      lowest_reference_tissue = tissue[which.min(mean_vst)][1],
      reference_tissue_means = paste0(
        tissue, "=", round(mean_vst, 2), collapse = "; "
      ),
      .groups = "drop"
    )

  tissue_pattern_gene_ids <- reference_tissue_signal |>
    dplyr::filter(
      n_reference_tissues >= 2,
      tissue_vst_range >= params$tissue_vst_range_threshold
    ) |>
    dplyr::pull(gene_id)

  phase_gene_evidence <- phase_gene_all |>
    dplyr::transmute(
      gene_id,
      phase_associated = n_phase_tissues > 0,
      phase_class,
      phase_consensus,
      phase_tissues,
      phase_mean_log2FC,
      phase_max_abs_log2FC
    )

  diet_infection_evidence <- phase_layered_sig |>
    dplyr::filter(scenario == params$analysis_scenario) |>
    dplyr::group_by(gene_id) |>
    dplyr::summarise(
      diet_infection_associated = TRUE,
      infection_associated = any(comparison_type == "Infection within diet"),
      diet_associated = any(comparison_type != "Infection within diet"),
      n_deg_contrasts = dplyr::n_distinct(comparison),
      deg_comparison_types = paste(sort(unique(comparison_type)),
                                   collapse = "; "),
      deg_comparisons = paste(sort(unique(comparison)), collapse = "; "),
      deg_directions = paste(sort(unique(direction)), collapse = "; "),
      max_abs_log2FC = max(abs(log2FoldChange), na.rm = TRUE),
      min_padj = min(padj, na.rm = TRUE),
      .groups = "drop"
    )

  partition_gene_ids <- Reduce(
    union,
    list(
      tissue_pattern_gene_ids,
      phase_gene_evidence$gene_id[phase_gene_evidence$phase_associated],
      diet_infection_evidence$gene_id
    )
  )

  source_partition_levels <- c(
    "Tissue reference pattern only",
    "Phase signature only",
    "Diet/infection DEG only",
    "Tissue + phase",
    "Tissue + diet/infection",
    "Phase + diet/infection",
    "Tissue + phase + diet/infection"
  )

  gene_source_partition <- tibble::tibble(gene_id = partition_gene_ids) |>
    dplyr::left_join(reference_tissue_signal, by = "gene_id") |>
    dplyr::left_join(phase_gene_evidence, by = "gene_id") |>
    dplyr::left_join(diet_infection_evidence, by = "gene_id") |>
    dplyr::left_join(annotation, by = "gene_id") |>
    dplyr::mutate(
      tissue_associated = gene_id %in% tissue_pattern_gene_ids,
      phase_associated = dplyr::coalesce(phase_associated, FALSE),
      diet_infection_associated =
        dplyr::coalesce(diet_infection_associated, FALSE),
      infection_associated = dplyr::coalesce(infection_associated, FALSE),
      diet_associated = dplyr::coalesce(diet_associated, FALSE),
      source_class = dplyr::case_when(
        tissue_associated & !phase_associated & !diet_infection_associated ~
          "Tissue reference pattern only",
        !tissue_associated & phase_associated & !diet_infection_associated ~
          "Phase signature only",
        !tissue_associated & !phase_associated & diet_infection_associated ~
          "Diet/infection DEG only",
        tissue_associated & phase_associated & !diet_infection_associated ~
          "Tissue + phase",
        tissue_associated & !phase_associated & diet_infection_associated ~
          "Tissue + diet/infection",
        !tissue_associated & phase_associated & diet_infection_associated ~
          "Phase + diet/infection",
        tissue_associated & phase_associated & diet_infection_associated ~
          "Tissue + phase + diet/infection",
        TRUE ~ "No selected evidence"
      ),
      source_class = factor(source_class, levels = source_partition_levels),
      Preferred_name = dplyr::coalesce(Preferred_name, "unannotated"),
      Description = dplyr::coalesce(Description, "no description"),
      priority_score = dplyr::coalesce(n_deg_contrasts, 0) * 10 +
        dplyr::coalesce(tissue_vst_range, 0) +
        dplyr::coalesce(phase_max_abs_log2FC, 0) +
        dplyr::coalesce(safe_neglog10(min_padj), 0)
    ) |>
    dplyr::arrange(source_class, dplyr::desc(priority_score), gene_id)

  source_partition_counts <- gene_source_partition |>
    dplyr::count(source_class, name = "n_genes") |>

Interpretation guide

The most useful output is not the raw overlap count alone, but the direction of the overlap:

  • Numerator more gregarious-like means the numerator condition moves in the same direction as crowded/gregarious animals in the independent phase dataset.
  • Numerator more solitarious-like means the numerator condition moves opposite to the crowded/gregarious signature, which is more consistent with an isolated/solitarious-like expression direction.
  • Mixed head/thorax phase signal means the gene is phase responsive, but the head and thorax reference signatures disagree in sign.

These genes should not be discarded automatically. Instead, they should be highlighted in candidate-gene interpretation as possible density-state sensitive genes that could confound or mediate diet and infection effects.
