Purpose

This page records the input files, sample labels, experimental groups, and basic count-table checks used by downstream analyses. It is meant to be knitted before running DESeq2 pages so mismatched sample names or incomplete metadata are caught early.

counts <- read.csv(counts_file, check.names = FALSE)
gene_ids <- counts[[1]]
count_matrix <- counts[, -1, drop = FALSE]
rownames(count_matrix) <- gene_ids

# Count columns include file prefixes; convert them back to metadata labels.
count_labels <- sub("_MERGE$", "", sub("^mehreen_", "", colnames(count_matrix)))
metadata_labels <- metadata$label

sample_match <- data.frame(
  count_column = colnames(count_matrix),
  inferred_label = count_labels,
  found_in_metadata = count_labels %in% metadata_labels
