Purpose

This page keeps the eggNOG-based functional enrichment with the differential expression results. It uses the latest DESeq2 result tables on disk, lifts eggNOG protein annotations onto the LOC... gene IDs in the count matrix, and tests infection DEG sets for enriched GO terms and KEGG pathways.

The key point is that this is a custom enrichment workflow for this reference: the foregrounds are current DEG sets, the background is the annotated gene universe in this count matrix, and GO names/ontologies come from GO.db.

The first enrichment section keeps the infection response non-directional, which asks what functions are over-represented among infection DEGs regardless of whether genes increase or decrease. A later section adds the requested direction-aware contrasts: infected-versus-control within each diet, diet contrasts among controls only, and diet contrasts among infected samples only.

Build eggNOG annotation maps

eggNOG reports annotations at the protein-query level, while the count matrix uses LOC... gene IDs. The GFF provides the bridge from CDS protein IDs to gene IDs. Only genes present in the count matrix are kept in the enrichment background.

read_emapper <- function(path) {
  cols <- c(
    "query", "seed_ortholog", "evalue", "score", "eggNOG_OGs",
    "max_annot_lvl", "COG_category", "Description", "Preferred_name",
    "GOs", "EC", "KEGG_ko", "KEGG_Pathway", "KEGG_Module",
    "KEGG_Reaction", "KEGG_rclass", "BRITE", "KEGG_TC", "CAZy",
    "BiGG_Reaction", "PFAMs"
  )
  tab <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", fill = TRUE)
  names(tab) <- cols[seq_len(ncol(tab))]
  tab |>
    dplyr::mutate(protein_id = stringr::str_extract(query, "XP_[0-9]+[.][0-9]+"))
}

parse_cds_gene_bridge <- function(path) {
  gff <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", stringsAsFactors = FALSE)
  names(gff)[1:9] <- c("seqid", "source", "type", "start", "end", "score",
                       "strand", "phase", "attributes")
  gff |>
    dplyr::filter(type == "CDS") |>
    dplyr::transmute(
      protein_id = stringr::str_match(attributes, "protein_id=([^;]+)")[, 2],
      gene_id = stringr::str_match(attributes, "gene=([^;]+)")[, 2]
    ) |>
    dplyr::filter(!is.na(protein_id), !is.na(gene_id)) |>
    dplyr::distinct()
}

get_kegg_pathway_names <- function(ids) {
  ids <- unique(ids)
  resolved <- setNames(rep(NA_character_, length(ids)), ids)

  if (requireNamespace("KEGGREST", quietly = TRUE)) {
    kegg_list <- tryCatch(
      KEGGREST::keggList("pathway", "ko"),
      error = function(e) character()
    )
    if (length(kegg_list) > 0) {
      live_ids <- sub("^path:", "", names(kegg_list))
      live_names <- sub(" - Reference pathway$", "", unname(kegg_list))
      live_names <- sub(" - Homo sapiens \\(human\\)$", "", live_names)
      resolved[intersect(ids, live_ids)] <- live_names[match(intersect(ids, live_ids),
                                                             live_ids)]
    }
  }

  fallback_ids <- intersect(ids, names(kegg_fallback_names))
  resolved[fallback_ids] <- ifelse(
    is.na(resolved[fallback_ids]),
    unname(kegg_fallback_names[fallback_ids]),
    resolved[fallback_ids]
  )
  resolved[is.na(resolved)] <- ids[is.na(resolved)]
  resolved
}

Collect current infection DEG sets

The enrichment foregrounds are built from infected-versus-control DEGs without splitting genes by direction. This keeps the infection-response question simple: which functional categories are over-represented among infection-responsive genes in general and within each diet. The newest file for each within-diet infection result table is used so older workflowR runs do not duplicate a contrast.

latest_result_files <- function(pattern) {
  files <- list.files(results_root, pattern = pattern, recursive = TRUE,
                      full.names = TRUE)
  if (length(files) == 0) {
    return(data.frame(file = character(), short_name = character(),
                      modified = as.POSIXct(character())))
  }
  info <- data.frame(
    file = files,
    short_name = basename(files),
    modified = file.info(files)$mtime,
    stringsAsFactors = FALSE
  )
  info <- info[order(info$short_name, -as.numeric(info$modified)), ]
  info[!duplicated(info$short_name), ]
}

read_infection_deg_set <- function(path) {
  tab <- read.csv(path, check.names = FALSE)
  gene_col <- intersect(c("gene_id", "gene", "GeneID"), names(tab))[1]
  if (is.na(gene_col) || !"padj" %in% names(tab) ||
      !"log2FoldChange" %in% names(tab)) return(NULL)
  sig <- !is.na(tab$padj) &
    tab$padj < params$alpha &
    abs(tab$log2FoldChange) >= params$lfc_threshold
  unique(as.character(tab[[gene_col]][sig]))
}

latest_files_by_name <- function(pattern) {
  files <- list.files(results_root, pattern = pattern, recursive = TRUE,
                      full.names = TRUE)
  if (length(files) == 0) {
    return(data.frame(file = character(), short_name = character(),
                      modified = as.POSIXct(character())))
  }
  info <- data.frame(
    file = files,
    short_name = basename(files),
    modified = file.info(files)$mtime,
    stringsAsFactors = FALSE
  )
  info <- info[order(info$short_name, -as.numeric(info$modified)), ]
  info[!duplicated(info$short_name), ]
}

add_directional_sets <- function(tab, source_file, set_prefix, family,
                                 comparison, up_label, down_label,
                                 up_plot_label, down_plot_label) {
  if (is.null(tab)) return(NULL)

  up_genes <- unique(tab$gene_id_for_set[
    tab$significant_for_set & tab$log2FoldChange > 0
  ])
  down_genes <- unique(tab$gene_id_for_set[
    tab$significant_for_set & tab$log2FoldChange < 0
  ])

  list(
    sets = list(
      stats::setNames(list(up_genes), paste0(set_prefix, "_up")),
      stats::setNames(list(down_genes), paste0(set_prefix, "_down"))
    ) |>
      unlist(recursive = FALSE),
    records = list(
      make_directional_record(
        paste0(set_prefix, "_up"), family, comparison, "Upregulated",
        up_plot_label, up_genes, source_file
      ),
      make_directional_record(
        paste0(set_prefix, "_down"), family, comparison, "Downregulated",
        down_plot_label, down_genes, source_file
      )
    )
  )
}

Fat body immune and storage candidates

The fat body is central to immune defense, detoxification, lipid transport, and nutrient storage. This table therefore flags significant infection-responsive genes whose eggNOG annotation contains immune, storage-protein, lipid-transport, endocrine-nutrition, or detoxification language. This is a transparent keyword screen to help prioritize genes for biological interpretation; it does not replace the GO or KEGG enrichment tests.

classify_fat_body_candidate <- function(text) {
  text <- stringr::str_to_lower(text)
  immune <- stringr::str_detect(
    text,
    "immune|defen[cs]e|defensin|attacin|cecropin|gloverin|lysozyme|toll|imd|tnf|jak-stat|peptidoglycan|antimicrobial|phenoloxidase|prophenoloxidase|serpin|thioester|transferrin|lectin|phagosome"
  )
  storage <- stringr::str_detect(
    text,
    "storage protein|hexamerin|arylphorin|hemocyanin|vitellogenin|lipophorin|apolipophorin|nutrient reservoir"
  )
  detox <- stringr::str_detect(
    text,
    "detox|glutathione|cytochrome p450|p450|xenobiotic|oxidative stress|peroxidase|superoxide|catalase"
  )
  endocrine_nutrition <- stringr::str_detect(
    text,
    "juvenile hormone|insulin|glucagon|adipocytokine|ppar|bile secretion|mineral absorption|fat digestion|glycerolipid|amino acid|lipid transport"
  )

  dplyr::case_when(
    immune & storage ~ "Immune + storage/lipid transport",
    storage ~ "Storage / lipid transport",
    immune ~ "Immune / defense",
    detox ~ "Detoxification / oxidative stress",
    endocrine_nutrition ~ "Endocrine / nutrient metabolism",
    TRUE ~ NA_character_
  )
}

run_enrichment <- function(genes, term2gene, term2name, universe) {
  genes <- unique(genes[genes %in% universe])
  if (length(genes) < 5) return(NULL)
  clusterProfiler::enricher(
    gene = genes,
    universe = universe,
    TERM2GENE = term2gene,
    TERM2NAME = term2name[, c("term", "name")],
    pvalueCutoff = 0.1,
    pAdjustMethod = "BH",
    qvalueCutoff = 0.2
  )
}

enrich_to_df <- function(result_list, database) {
  pieces <- lapply(names(result_list), function(nm) {
    res <- result_list[[nm]]

    if (nrow(df) == 0) return(NULL)
    df$set <- nm
    df$database <- database
    df
  })
  dplyr::bind_rows(pieces)
