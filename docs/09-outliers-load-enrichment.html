Current sample and outlier status

Outliers are handled explicitly because this experiment has a modest number of replicates per diet-by-treatment group. A sample that separates strongly in PCA, sample-distance heatmaps, or library-level QC can pull a group mean away from the other biological replicates and inflate or hide DEG calls. The no-outlier metadata file is therefore treated as a sensitivity dataset: it does not erase the full dataset, but it lets the analysis ask whether the main conclusions depend on the strongest QC outliers.

Build annotation maps from local eggNOG and GFF

The count matrix uses LOC... gene IDs, whereas the eggNOG file is annotated at the protein-query level. This section uses the local GFF file as a bridge from protein IDs to gene IDs, then keeps only annotations that can be lifted onto genes present in the count matrix. That makes GO and KEGG enrichment compatible with the DESeq2 result tables.

read_emapper <- function(path) {
  cols <- c(
    "query", "seed_ortholog", "evalue", "score", "eggNOG_OGs",
    "max_annot_lvl", "COG_category", "Description", "Preferred_name",
    "GOs", "EC", "KEGG_ko", "KEGG_Pathway", "KEGG_Module",
    "KEGG_Reaction", "KEGG_rclass", "BRITE", "KEGG_TC", "CAZy",
    "BiGG_Reaction", "PFAMs"
  )
  tab <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", fill = TRUE)
  names(tab) <- cols[seq_len(ncol(tab))]
  tab |>
    dplyr::mutate(protein_id = stringr::str_extract(query, "XP_[0-9]+[.][0-9]+"))
}

parse_cds_gene_bridge <- function(path) {
  gff <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", stringsAsFactors = FALSE)
  names(gff)[1:9] <- c("seqid", "source", "type", "start", "end", "score",
                       "strand", "phase", "attributes")
  gff |>
    dplyr::filter(type == "CDS") |>
    dplyr::transmute(
      protein_id = stringr::str_match(attributes, "protein_id=([^;]+)")[, 2],
      gene_id = stringr::str_match(attributes, "gene=([^;]+)")[, 2]
    ) |>
    dplyr::filter(!is.na(protein_id), !is.na(gene_id)) |>
    dplyr::distinct()
}

parse_gene_biotypes <- function(path) {
  gff <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", stringsAsFactors = FALSE)
  names(gff)[1:9] <- c("seqid", "source", "type", "start", "end", "score",
                       "strand", "phase", "attributes")
  gff |>
    dplyr::filter(type == "gene") |>
    dplyr::transmute(
      gene_id = stringr::str_match(attributes, "gene=([^;]+)")[, 2],
      gene_biotype = stringr::str_match(attributes, "gene_biotype=([^;]+)")[, 2],
      gene_description = stringr::str_match(attributes, "description=([^;]+)")[, 2]
    ) |>
    dplyr::filter(!is.na(gene_id)) |>
    dplyr::distinct(gene_id, .keep_all = TRUE)
}

read_deg_table <- function(path) {
  tab <- read.csv(path, check.names = FALSE)
  gene_col <- intersect(c("gene_id", "gene", "GeneID"), names(tab))[1]
  if (is.na(gene_col) || !"padj" %in% names(tab) ||
      !"log2FoldChange" %in% names(tab)) return(NULL)
  tab |>
    dplyr::transmute(
      gene_id = as.character(.data[[gene_col]]),
      log2FoldChange = log2FoldChange,
      padj = padj,
      source = tools::file_path_sans_ext(basename(path)),
      direction = dplyr::case_when(
        !is.na(padj) & padj < params$alpha &
          log2FoldChange >= params$lfc_threshold ~ "UP",
        !is.na(padj) & padj < params$alpha &
          log2FoldChange <= -params$lfc_threshold ~ "DOWN",
        !is.na(padj) & padj < params$alpha &
          abs(log2FoldChange) >= params$lfc_threshold ~ "ALL",
        TRUE ~ "NOT_SIGNIFICANT"
      )
    )
}

Coding and long non-coding DEG classes

This section mirrors the AGY DEG magnitude figure. It asks whether each DEG set is mostly protein-coding, long non-coding RNA, or another/unknown gene class. Gene class comes from gene_biotype in the local GFF, while DEG direction comes from the current DESeq2 result tables.

classify_gene <- function(x) {
  dplyr::case_when(
    x == "protein_coding" ~ "Coding genes",
    x == "lncRNA" ~ "Long non-coding RNA",
    is.na(x) ~ "Other / unknown",
    TRUE ~ "Other non-coding"
  )
}

deg_biotype <- deg_long |>
  dplyr::filter(direction %in% c("UP", "DOWN")) |>
  dplyr::left_join(gene_biotypes, by = "gene_id") |>
  dplyr::mutate(
    Gene_class = classify_gene(gene_biotype),
    Direction = dplyr::recode(direction,
                              UP = "Higher in numerator/infected",
                              DOWN = "Higher in denominator/control"),
    signed_direction = ifelse(direction == "UP", 1, -1),
    source_label = stringr::str_replace_all(source, "_", " ")
  ) |>
  dplyr::distinct(source, source_label, direction, Direction,
                  gene_id, Gene_class, .keep_all = TRUE)

GO and KEGG enrichment

GO and KEGG enrichment asks whether a DEG set contains more genes from a functional term than expected from the annotated gene universe. The universe is restricted to genes that are both present in the count matrix and represented in the corresponding annotation map, so the background matches what this dataset could actually test.

run_enrichment <- function(genes, term2gene, term2name, universe) {
  genes <- unique(genes[genes %in% universe])
  if (length(genes) < 5) return(NULL)
  clusterProfiler::enricher(
    gene = genes,
    universe = universe,
    TERM2GENE = term2gene,
    TERM2NAME = term2name[, c("term", "name")],
    pvalueCutoff = 0.1,
    pAdjustMethod = "BH",
    qvalueCutoff = 0.2
  )
}

enrichment_sets <- split(deg_long$gene_id, deg_long$source_direction)

enrich_to_df <- function(result_list, database) {
  pieces <- lapply(names(result_list), function(nm) {
    res <- result_list[[nm]]
    if (is.null(res)) return(NULL)
    df <- as.data.frame(res)
    if (nrow(df) == 0) return(NULL)
    df$set <- nm
    df$database <- database
    df
  })
  dplyr::bind_rows(pieces)
}
