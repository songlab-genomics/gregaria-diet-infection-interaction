Purpose

The biological goal is to identify genes that differ among diets, infection status, and diet-specific infection responses. This page summarizes current sample inclusion, lifts local eggNOG annotations onto the LOC... gene IDs used by the count matrix, and runs GO and KEGG pathway enrichment on the current DEG sets.

Current sample and outlier status

Outliers are handled explicitly because this experiment has a modest number of replicates per diet-by-treatment group. A sample that separates strongly in PCA, sample-distance heatmaps, or library-level QC can pull a group mean away from the other biological replicates and inflate or hide DEG calls. The no-outlier metadata file is therefore treated as a sensitivity dataset: it does not erase the full dataset, but it lets the analysis ask whether the main conclusions depend on the two strongest QC outliers.

Build annotation maps from local eggNOG and GFF

The count matrix uses LOC... gene IDs, whereas the eggNOG file is annotated at the protein-query level. This section uses the local GFF file as a bridge from protein IDs to gene IDs, then keeps only annotations that can be lifted onto genes present in the count matrix. That makes GO and KEGG enrichment compatible with the DESeq2 result tables.

read_emapper <- function(path) {
  cols <- c(
    "query", "seed_ortholog", "evalue", "score", "eggNOG_OGs",
    "max_annot_lvl", "COG_category", "Description", "Preferred_name",
    "GOs", "EC", "KEGG_ko", "KEGG_Pathway", "KEGG_Module",
    "KEGG_Reaction", "KEGG_rclass", "BRITE", "KEGG_TC", "CAZy",
    "BiGG_Reaction", "PFAMs"
  )
  tab <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", fill = TRUE)
  names(tab) <- cols[seq_len(ncol(tab))]
  tab |>
    dplyr::mutate(protein_id = stringr::str_extract(query, "XP_[0-9]+[.][0-9]+"))
}

parse_cds_gene_bridge <- function(path) {
  gff <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", stringsAsFactors = FALSE)
  names(gff)[1:9] <- c("seqid", "source", "type", "start", "end", "score",
                       "strand", "phase", "attributes")
  gff |>
    dplyr::filter(type == "CDS") |>
    dplyr::transmute(
      protein_id = stringr::str_match(attributes, "protein_id=([^;]+)")[, 2],
      gene_id = stringr::str_match(attributes, "gene=([^;]+)")[, 2]
    ) |>
    dplyr::filter(!is.na(protein_id), !is.na(gene_id)) |>
    dplyr::distinct()
}

Enrichment input DEG sets

For enrichment, each DEG table is split by source and direction. Direction-aware sets are helpful because genes higher in infected samples and genes higher in controls can point to different biological processes, even when they come from the same statistical contrast.

result_files <- list.files(file.path(project_dir, params$latest_results_root),

read_deg_table <- function(path) {
  tab <- read.csv(path, check.names = FALSE)
  gene_col <- intersect(c("gene_id", "gene", "GeneID"), names(tab))[1]
  if (is.na(gene_col) || !"padj" %in% names(tab) ||
      !"log2FoldChange" %in% names(tab)) return(NULL)
  tab |>
    dplyr::transmute(
      gene_id = as.character(.data[[gene_col]]),
      log2FoldChange = log2FoldChange,
      padj = padj,
      source = tools::file_path_sans_ext(basename(path)),
      direction = dplyr::case_when(
        !is.na(padj) & padj < params$alpha &
          log2FoldChange >= params$lfc_threshold ~ "UP",
        !is.na(padj) & padj < params$alpha &
          log2FoldChange <= -params$lfc_threshold ~ "DOWN",
        !is.na(padj) & padj < params$alpha &
          abs(log2FoldChange) >= params$lfc_threshold ~ "ALL",
        TRUE ~ "NOT_SIGNIFICANT"
      )
    )
}

GO and KEGG enrichment

GO and KEGG enrichment asks whether a DEG set contains more genes from a functional term than expected from the annotated gene universe. The universe is restricted to genes that are both present in the count matrix and represented in the corresponding annotation map, so the background matches what this dataset could actually test.

run_enrichment <- function(genes, term2gene, term2name, universe) {
  genes <- unique(genes[genes %in% universe])
  if (length(genes) < 5) return(NULL)
  clusterProfiler::enricher(
    gene = genes,
    universe = universe,
    TERM2GENE = term2gene,
    TERM2NAME = term2name[, c("term", "name")],
    pvalueCutoff = 0.1,
    pAdjustMethod = "BH",
    qvalueCutoff = 0.2
  )
}

enrichment_sets <- split(deg_long$gene_id, deg_long$source_direction)

universe_kegg <- unique(term2gene_kegg_pathway$gene)

enrich_to_df <- function(result_list, database) {
  pieces <- lapply(names(result_list), function(nm) {
    res <- result_list[[nm]]
    if (is.null(res)) return(NULL)
    df <- as.data.frame(res)
    if (nrow(df) == 0) return(NULL)
    df$set <- nm
    df$database <- database
    df
  })
  dplyr::bind_rows(pieces)
}

Term overlap across DEG sets

This final heatmap highlights functional terms that recur in multiple DEG sets. Shared terms can point to processes that are robust across contrast definitions, whereas terms seen in only one set may be more diet-, infection-, or direction-specific.

if (nrow(all_enrichment) > 0) {

    dplyr::filter(p.adjust < 0.1) |>
    dplyr::distinct(database, ID, Description, set) |>
    dplyr::add_count(database, ID, name = "n_sets") |>
    dplyr::filter(n_sets >= 2)

  write.csv(term_overlap, file.path(out_dir, "enriched_terms_seen_in_multiple_sets.csv"),
            row.names = FALSE)

  if (nrow(term_overlap) > 0) {
    top_overlap_terms <- term_overlap |>
      dplyr::distinct(database, ID, Description, n_sets) |>
      dplyr::arrange(dplyr::desc(n_sets)) |>
      head(40)

    overlap_plot_df <- term_overlap |>
      dplyr::semi_join(top_overlap_terms, by = c("database", "ID", "Description", "n_sets")) |>
      dplyr::mutate(
        term_label = stringr::str_trunc(Description, 60),
        set = stringr::str_replace_all(set, "_", " ")
      )

    p_overlap <- ggplot2::ggplot(overlap_plot_df,
                                 ggplot2::aes(x = set, y = term_label)) +
      ggplot2::geom_tile(fill = "#2b8cbe") +

      ggplot2::theme(axis.text.x = ggplot2::element_text(angle = 45, hjust = 1)) +
