Purpose

This page tests infected versus control samples separately within each diet. This is a simple and readable complement to the full interaction model.

The reason to split by diet is biological and statistical: infection may not produce the same transcriptional response under every nutritional condition. The full interaction model tests whether diet modifies the infection effect; this page then turns that question into three direct infected-control comparisons that are easier to inspect, plot, and interpret.

UMAP can be useful for pattern discovery, but it does not give the same direct variance accounting and can change with parameter choices. Here the small points are individual libraries, the larger points are diet-by-treatment centroids, and the pale condition polygons connect the outermost samples from each diet-by-treatment group in the first two PCs.

Run one infected-control test per diet

For each diet, the code keeps only samples from that diet and fits a simple ~ Treatment DESeq2 model. This avoids averaging the infection effect across diets. A positive log2 fold-change means higher expression in infected samples; a negative log2 fold-change means higher expression in controls. Genes are called significant only when they pass both the adjusted p-value threshold and the absolute log2 fold-change threshold.

diet_results <- list()

for (diet_level in levels(droplevels(metadata$Diet))) {
  message("Running infection contrast for diet ", diet_level)

  sample_keep <- metadata$Diet == diet_level
  metadata_sub <- droplevels(metadata[sample_keep, ])
  counts_sub <- counts[, sample_keep]
  gene_keep <- rowSums(counts_sub) >= params$min_total_count

  dds <- DESeq2::DESeqDataSetFromMatrix(
    countData = counts_sub[gene_keep, ],
    colData = metadata_sub,
    design = ~ Treatment
  )
  dds <- DESeq2::DESeq(dds)

  res <- DESeq2::results(dds,
                         contrast = c("Treatment", "Infected", "Control"),
                         alpha = params$alpha)
  res_df <- as.data.frame(res)
  res_df$gene_id <- rownames(res_df)
  res_df$diet <- diet_level
  res_df$significant <- !is.na(res_df$padj) &
    res_df$padj < params$alpha &
    abs(res_df$log2FoldChange) >= params$lfc_threshold

  diet_results[[diet_level]] <- res_df
  write.csv(res_df, file.path(out_dir,
                              paste0("DESeq2_infected_vs_control_diet_",
                                     diet_level, ".csv")),
            row.names = FALSE)
  writeLines(res_df$gene_id[res_df$significant],
             file.path(out_dir,
                       paste0("significant_genes_infection_diet_",
                              diet_level, ".txt")))

  print(EnhancedVolcano::EnhancedVolcano(
    res_df,
    lab = res_df$gene_id,
    x = "log2FoldChange",
    y = "padj",
    pCutoff = params$alpha,
    FCcutoff = params$lfc_threshold,
    title = paste("Infected vs control in diet", diet_level)
  ))
}

all_infection_results <- do.call(rbind, diet_results)

Diet-specific DEG PCA

The PCA above uses all filtered genes and all samples, so it describes global expression structure. This section asks a narrower question for each diet: if we look only at the genes that are differentially expressed between infected and control samples within that diet, do the samples separate by treatment? This is closer to the DEG result itself, and it avoids using genes from another diet to explain the infected-control pattern.

make_deg_pca <- function(diet_level) {
  res <- diet_results[[diet_level]]
  deg_genes <- res$gene_id[res$significant & !is.na(res$gene_id)]
  deg_genes <- intersect(deg_genes, rownames(vst_mat))
  sample_keep <- metadata$Diet == diet_level
  metadata_sub <- droplevels(metadata[sample_keep, ])

  if (length(deg_genes) < 3 || nrow(metadata_sub) < 4) {
    return(NULL)
  }

  mat <- vst_mat[deg_genes, metadata_sub$label, drop = FALSE]
  keep_gene <- apply(mat, 1, var) > 0
  mat <- mat[keep_gene, , drop = FALSE]
  if (nrow(mat) < 3) return(NULL)

  pca_local <- prcomp(t(mat), scale. = FALSE)
  percent_local <- round(100 * (pca_local$sdev^2 / sum(pca_local$sdev^2)), 1)
  scores <- data.frame(
    label = rownames(pca_local$x),
    PC1 = pca_local$x[, "PC1"],
    PC2 = pca_local$x[, "PC2"],
    Diet = metadata_sub$Diet,
    Treatment = metadata_sub$Treatment,
    check.names = FALSE
  )

  centroids <- aggregate(cbind(PC1, PC2) ~ Treatment, scores, mean)
  names(centroids)[names(centroids) == "PC1"] <- "centroid_PC1"
  names(centroids)[names(centroids) == "PC2"] <- "centroid_PC2"

  treatment_hulls <- do.call(rbind, lapply(split(scores, scores$Treatment),
                                           make_condition_hull))

  p <- ggplot2::ggplot(scores, ggplot2::aes(PC1, PC2)) +
    ggplot2::geom_polygon(
      data = treatment_hulls,
      ggplot2::aes(x = PC1, y = PC2, fill = Treatment, group = Treatment),
      alpha = 0.22,
      color = NA,
      inherit.aes = FALSE
    ) +
    ggplot2::geom_point(ggplot2::aes(color = Treatment, shape = Treatment),
                        size = 3.2, alpha = 0.9) +
    ggplot2::geom_point(
      data = centroids,
      ggplot2::aes(x = centroid_PC1, y = centroid_PC2, color = Treatment),
      shape = 21,
      fill = "white",
      stroke = 0.8,
      size = 5,
      inherit.aes = FALSE
    ) +
    ggrepel::geom_label_repel(
      data = centroids,
      ggplot2::aes(x = centroid_PC1, y = centroid_PC2, label = Treatment),
      color = "grey15",
      fill = "white",
      label.size = 0.18,
      size = 3.2,
      min.segment.length = 0,
      inherit.aes = FALSE
    ) +
    ggplot2::scale_color_manual(values = treatment_colors) +
    ggplot2::scale_fill_manual(values = treatment_fills) +
    ggplot2::theme_classic() +
    ggplot2::theme(legend.position = "bottom") +
    ggplot2::labs(
      title = paste("Diet", diet_level, "PCA using infection DEGs only"),
      subtitle = paste(length(deg_genes), "significant genes"),
      x = paste0("PC1: ", percent_local[1], "% variance"),
      y = paste0("PC2: ", percent_local[2], "% variance"),
      color = "Treatment",
      fill = "Treatment",
      shape = "Treatment"
    )

  ggsave(file.path(out_dir,
                   paste0("diet_", diet_level, "_infection_deg_pca.png")),
         p, width = 7.2, height = 5.2, dpi = 300)
  write.csv(scores, file.path(out_dir,
                              paste0("diet_", diet_level,
                                     "_infection_deg_pca_scores.csv")),
            row.names = FALSE)
  write.csv(centroids, file.path(out_dir,
                                 paste0("diet_", diet_level,
                                        "_infection_deg_pca_centroids.csv")),
            row.names = FALSE)
  p
}

for (diet_level in levels(droplevels(metadata$Diet))) {

    cat("Diet", diet_level,

    print(p_local)

Curated DEG VST heatmap

The full VST heatmap is a broad QC view. This heatmap is narrower: it selects the strongest infected-control DEGs from each diet and direction, then adds significant DEGs with eggNOG descriptions matching immune or nutrition-related keywords. This follows the same idea as the AGY candidate heatmaps: keep the statistical DEG signal visible, but annotate genes that are biologically relevant to the manuscript question. Samples are ordered by infection status first and then diet instead of clustered, so the infected profiles appear together and diet-specific substructure can be compared directly. Genes are ordered first by how broadly they are detected as DEGs across diets, and then by their functional theme.

read_emapper <- function(path) {
  cols <- c(
    "query", "seed_ortholog", "evalue", "score", "eggNOG_OGs",
    "max_annot_lvl", "COG_category", "Description", "Preferred_name",
    "GOs", "EC", "KEGG_ko", "KEGG_Pathway", "KEGG_Module",
    "KEGG_Reaction", "KEGG_rclass", "BRITE", "KEGG_TC", "CAZy",
    "BiGG_Reaction", "PFAMs"
  )
  tab <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", fill = TRUE)
  names(tab) <- cols[seq_len(ncol(tab))]
  tab$protein_id <- sub(".*(XP_[0-9]+[.][0-9]+).*", "\\1", tab$query)
  tab
}

parse_cds_gene_bridge <- function(path) {
  gff <- read.delim(path, comment.char = "#", header = FALSE, sep = "\t",
                    quote = "", stringsAsFactors = FALSE)
  names(gff)[1:9] <- c("seqid", "source", "type", "start", "end", "score",
                       "strand", "phase", "attributes")
  cds <- gff[gff$type == "CDS", ]
  data.frame(
    protein_id = sub(".*protein_id=([^;]+).*", "\\1", cds$attributes),
    gene_id = sub(".*gene=([^;]+).*", "\\1", cds$attributes),
    stringsAsFactors = FALSE
  ) |>
    dplyr::filter(grepl("^XP_", protein_id), grepl("^LOC", gene_id)) |>
    dplyr::distinct()
}
