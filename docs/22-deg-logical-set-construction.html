Purpose

This page separates the DEG questions before comparing them. The fat body experiment remains the primary dataset for diet and infection, but the reference data are now modeled independently instead of being forced to pass through the fat body DEG list first.

The logic is:

  1. Fit DESeq2 on the reference tissues only, using all copied fat body, head, and thorax reference samples regardless of phase. This gives tissue-associated DEGs.
  2. Keep head and thorax phase-associated DEGs as the phase reference set. This asks whether a fat body diet/infection DEG is also phase-related in another sampled tissue.
  3. Keep fat body experiment DEGs as separate sets for diet among controls and infection within each diet.
  4. Only after those independent sets exist, compare them with Venn/UpSet plots and heatmaps.

Here, DEG categories are set-membership categories, not causal labels. A gene can therefore belong to more than one biological hypothesis class. For example, a fat body infection-response DEG can also be a tissue DEG or a phase DEG in the reference data.

The experiment fat body and the reference fat body are kept as separate sample blocks in PCA and heatmap annotations because they come from different experimental sources, even though both are fat body tissue.

read_metadata <- function(path) {
  tab <- read.delim(path, check.names = FALSE)
  tab <- tab[, !is.na(names(tab)) & nzchar(names(tab)), drop = FALSE]
  tab$label <- as.character(tab$label)
  tab$Diet <- factor(tab$Diet, levels = c("33", "50", "83"))
  tab$Treatment <- factor(tab$Treatment, levels = c("Infected", "Control"))
  tab
}

read_phase_table <- function(path, tissue) {
  tab <- read.csv(path, check.names = FALSE)
  bad_names <- is.na(names(tab)) | names(tab) == ""
  names(tab)[bad_names] <- paste0("row_id", seq_len(sum(bad_names)))
  if (!"GeneID" %in% names(tab)) {
    tab$GeneID <- tab[[1]]
  }
  tab |>
    dplyr::transmute(
      gene_id = as.character(GeneID),
      tissue = tissue,
      baseMean = baseMean,
      log2FoldChange = log2FoldChange,
      pvalue = pvalue,
      padj = padj,
      significant = !is.na(padj) &
        padj < params$phase_alpha &
        abs(log2FoldChange) >= params$phase_lfc_threshold,
      direction = dplyr::case_when(
        !significant ~ "Not DEG",
        log2FoldChange > 0 ~ "Gregarious-up",
        log2FoldChange < 0 ~ "Solitarious-up",
        TRUE ~ "Not DEG"
      )
    )
}

read_featurecounts_gene_counts <- function(path, sample_id) {
  tab <- read.delim(path, comment.char = "#", check.names = FALSE)
  if ("Geneid" %in% names(tab)) {
    count_col <- names(tab)[ncol(tab)]
    out <- tibble::tibble(
      gene_id = as.character(tab[["Geneid"]]),
      count = as.integer(tab[[count_col]])
    )
  } else {
    tab <- read.delim(path, comment.char = "#", header = FALSE,
                      check.names = FALSE)
    out <- tibble::tibble(
      gene_id = as.character(tab[[1]]),
      count = as.integer(tab[[ncol(tab)]])
    )
  }
  out <- out |>
    dplyr::filter(!is.na(gene_id), nzchar(gene_id)) |>
    dplyr::group_by(gene_id) |>
    dplyr::summarise(count = sum(count, na.rm = TRUE), .groups = "drop")
  names(out)[2] <- sample_id
  out
}

  files <- list.files(
    reference_dir,
    pattern = "^SGRE-(HEAD|THOX)-(CRD|ISO)-[0-9]+_MERGE_featurecounts[.]txt$",
    full.names = TRUE
  )
  parsed <- stringr::str_match(
    basename(files),
    "^SGRE-(HEAD|THOX)-(CRD|ISO)-([0-9]+)_MERGE_featurecounts[.]txt$"
  )
  tibble::tibble(
    sample_id = stringr::str_remove(basename(files), "_featurecounts[.]txt$"),
    count_file = files,
    tissue_code = parsed[, 2],
    rearing_code = parsed[, 3],
    replicate = as.integer(parsed[, 4])
  ) |>
    dplyr::filter(!is.na(tissue_code), !is.na(rearing_code)) |>
    dplyr::mutate(
      tissue = dplyr::case_when(
        tissue_code == "HEAD" ~ "Head",
        tissue_code == "THOX" ~ "Thorax",
        TRUE ~ tissue_code
      ),
      phase = dplyr::case_when(
        rearing_code == "CRD" ~ "Gregarious",
        rearing_code == "ISO" ~ "Solitarious",
        TRUE ~ rearing_code
      ),
      rearing_condition = dplyr::case_when(
        rearing_code == "CRD" ~ "Crowded",
        rearing_code == "ISO" ~ "Isolated",
        TRUE ~ rearing_code
      ),
      sample_label = paste0(tissue, "_", rearing_condition, "_", replicate),
      sample_block = paste(tissue, "phase reference")
    ) |>
    dplyr::arrange(tissue, phase, replicate)
}

run_pairwise_tissue_deseq <- function(count_mat, sample_metadata) {
  sample_metadata <- sample_metadata |>
    dplyr::mutate(
      phase = factor(phase, levels = c("Gregarious", "Solitarious")),
      tissue = factor(tissue, levels = c("Fat body", "Head", "Thorax"))
    )
  rownames(sample_metadata) <- sample_metadata$sample_id

      keep <- rowSums(count_mat) >= 10
      dds <- DESeq2::DESeqDataSetFromMatrix(
        countData = count_mat[keep, rownames(sample_metadata), drop = FALSE],
        colData = sample_metadata,
        design = ~ tissue
      )
  dds <- DESeq2::DESeq(dds, quiet = TRUE)

  contrasts <- tibble::tribble(
    ~comparison, ~numerator, ~denominator,
    "Fat body vs Head", "Fat body", "Head",
    "Fat body vs Thorax", "Fat body", "Thorax",
    "Head vs Thorax", "Head", "Thorax"
  )

  dplyr::bind_rows(lapply(seq_len(nrow(contrasts)), function(i) {
    res <- DESeq2::results(
      dds,
      contrast = c("tissue", contrasts$numerator[i], contrasts$denominator[i])
    )
    as.data.frame(res) |>
      tibble::rownames_to_column("gene_id") |>
      dplyr::mutate(
        comparison = contrasts$comparison[i],
            numerator = contrasts$numerator[i],
            denominator = contrasts$denominator[i],
            comparison_type = "Tissue contrast",
            model_formula = "~ tissue",
        significant = !is.na(padj) &
          padj < params$tissue_alpha &
          abs(log2FoldChange) >= params$tissue_lfc_threshold,
        direction = dplyr::case_when(
          !significant ~ "Not DEG",
          log2FoldChange > 0 ~ paste("Higher in", contrasts$numerator[i]),
          log2FoldChange < 0 ~ paste("Higher in", contrasts$denominator[i]),
          TRUE ~ "Not DEG"
        )
      )
  }))
}

      Treatment = factor(Treatment, levels = c("Control", "Infected"))
    )
  rownames(sample_metadata) <- sample_metadata$label

  keep <- rowSums(count_mat) >= 10
  dds <- DESeq2::DESeqDataSetFromMatrix(
    countData = count_mat[keep, rownames(sample_metadata), drop = FALSE],
    colData = sample_metadata,
    design = ~ Diet + Treatment
  )
  dds <- DESeq2::DESeq(dds, quiet = TRUE)
  res <- DESeq2::results(dds, contrast = c("Treatment", "Infected", "Control"))

  as.data.frame(res) |>
    tibble::rownames_to_column("gene_id") |>
    dplyr::mutate(
      scenario = scenario_label,
      comparison_type = "Pooled infection across diets",
      comparison = "All diets: infected vs control",
      numerator = "Infected",
      denominator = "Control",
      diet = "All",
      significant = !is.na(padj) &
        padj < params$alpha &
        abs(log2FoldChange) >= params$lfc_threshold,
      direction = dplyr::case_when(
        !significant ~ "Not DEG",
        log2FoldChange > 0 ~ "Upregulated",
        log2FoldChange < 0 ~ "Downregulated",
        TRUE ~ "Not DEG"
      ),
      comparison_family = "Pooled infection across diets"
    )
}

    tibble::rownames_to_column("gene_id") |>

      sample_label = factor(sample_label,

                              match(sample_order, sample_metadata$sample_id)
                            ]),
      gene_label = factor(gene_label, levels = rev(gene_table$gene_label))
    )
}
